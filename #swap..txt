#swap
def swap(x,y):
   print("before swap:",x)
   print("before swap:",y)
   x,y=y,x
   return x,y


#main rountine
a=int(input("enter a value"))
b=int(input("enter b value"))
a,b= swap(a,b)
print("after swapping becomes",a)
print("after swapping becomes",b)
