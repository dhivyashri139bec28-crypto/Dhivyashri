#USING RECURSIVE FUNCTION
#FIBONACCI SERIES
def fibo(n):
   if n<=0:
      return "invalid input,the input should be positive integer"
   elif n==1:
      return 0
   elif n==2:
      return 1
   else:
      return fibo(n-1)+fibo(n-2)

#main routine
n=int(input("enter the number"))
print("fibonacci series")
for i in range(1,n+1):
   print(fibo(i),end="")
